Trợ giúp & cài đặt

App này lưu dữ liệu trực tiếp vào Google Sheet — mọi nhân viên mở cùng 1 link đều đọc/ghi chung 1 danh sách, kể cả khi host trên GitHub Pages.

Cách kết nối Google Sheet (làm 1 lần)

1. Tạo 1 Google Sheet mới, dòng đầu đặt tên cột: ID, Ngay, Ten khach, SDT, Nhan vien, Ghi chu, Gia tien, Hinh thuc, CreatedAt.
2. Vào Tiện ích mở rộng → Apps Script, xoá code mẫu, dán đoạn code bên dưới.
3. Bấm Triển khai → Triển khai mới → Ứng dụng web. Chọn "Thực thi với quyền của: Tôi" và "Người có quyền truy cập: Bất kỳ ai", rồi triển khai.
4. Sao chép URL ứng dụng web, mở file index.html bằng trình soạn thảo, tìm dòng SHEET_API_URL ở đầu phần script và dán URL vào đó, rồi lưu lại trước khi tải lên GitHub Pages. 

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var rows = data.slice(1).filter(function(r){ return r[0]; });
  var result = rows.map(function(r){
    return {
      id: String(r[0]),
      ngay: (r[1] instanceof Date) ? Utilities.formatDate(r[1], Session.getScriptTimeZone(), "yyyy-MM-dd") : r[1],
      tenKhach: r[2], sdt: r[3], nhanVien: r[4],
      ghiChu: r[5], giaTien: r[6], hinhThuc: r[7], createdAt: r[8]
    };
  });
  return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var body = JSON.parse(e.postData.contents);
  if (body.action === 'add') {
    var d = body.entry;
    sheet.appendRow([d.id, d.ngay, d.tenKhach, d.sdt, d.nhanVien, d.ghiChu, d.giaTien, d.hinhThuc, d.createdAt]);
  } else if (body.action === 'update') {
    var d2 = body.entry;
    var data = sheet.getDataRange().getValues();
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(d2.id)) {
        sheet.getRange(i+1,1,1,9).setValues([[d2.id,d2.ngay,d2.tenKhach,d2.sdt,d2.nhanVien,d2.ghiChu,d2.giaTien,d2.hinhThuc,d2.createdAt]]);
        break;
      }
    }
  } else if (body.action === 'delete') {
    var data2 = sheet.getDataRange().getValues();
    for (var j = 1; j < data2.length; j++) {
      if (String(data2[j][0]) === String(body.id)) { sheet.deleteRow(j+1); break; }
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ok:true})).setMimeType(ContentService.MimeType.JSON);
}
